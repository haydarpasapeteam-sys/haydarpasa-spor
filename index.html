// Duyuruları JSON'dan Yükleyen Motor async function loadDuyurular() { const container = document.getElementById('duyuruContainer'); if (!container) return; try { // Tarayıcı önbelleğine takılmaması için zaman damgasıyla çağırıyoruz const response = await fetch('duyurular.json?v=' + Date.now()); if (!response.ok) throw new Error('Veri alınamadı: ' + response.status); const data = await response.json(); let duyurular = data.liste || []; if (duyurular.length === 0) { container.innerHTML = '
Henüz aktif duyuru bulunmuyor.
'; return; } // Yeni eklenen duyuruların (listenin sonundakilerin) en başta görünmesi için: duyurular = duyurular.slice().reverse(); container.innerHTML = duyurular.map(item => { let buttonsHtml = ''; if (item.form_linki && item.belge_linki) { buttonsHtml = `
Başvuru / Form İzin Belgesi
`; } else if (item.form_linki) { buttonsHtml = ` Başvuru / Form Linki `; } else if (item.belge_linki) { buttonsHtml = ` 📄 Veli İzin Belgesi `; } return `
${item.kategori || 'DUYURU'} ${item.tarih_saat || ''}

${item.baslik}

${item.aciklama || ''}

${buttonsHtml}
`; }).join(''); startAutoScroll(); } catch (err) { console.error('Duyuru yükleme hatası:', err); container.innerHTML = '
Duyurular yüklenirken bir sorun oluştu.
'; } }
